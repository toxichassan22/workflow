023553:       if (!text || !tenantSlidesData.length) return null;
023554:       const normalized = normalizeArabicDigits(text.trim().toLowerCase());
023555: 
023556:       // Pattern 1: Move last slide e.g. "انقل الشريحة الأخيرة لخامسة/رقم 4"
023557:       const lastMatch = normalized.match(/(?:انقل|حرك|غير|ضع|خلي)\s*(?:الشريحة|شريحة|سلايد)?\s*الأخيرة\s*(?:إلى|مكان|خليها|تكون|بعد|قبل|لتصبح)?\s*(?:الشريحة|شريحة|سلايد)?\s*(?:رقم\s*)?(\d+)/i);
023558:       if (lastMatch && lastMatch[1]) {

023560:         let toIdx = parseInt(lastMatch[1], 10) - 1;
023561:         if (fromIdx >= 0 && toIdx >= 0 && toIdx < tenantSlidesData.length && fromIdx !== toIdx) {
023562:           return { fromIdx, toIdx };
023563:         }
023564:       }
023565: 
023566:       // Pattern 2: Explicit numbers e.g. "انقل الشريحة 8 وخليها رقم 4" or "حرك 8 الى 4" or "ضع 8 بعد 3"

023576:           return { fromIdx, toIdx };
